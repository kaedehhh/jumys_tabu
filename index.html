ҰСЫНЫС ЖҮЙЕСІ: СӘЙКЕСТІК + СҰРЫПТАУ # ===================================================================== # Сұрыптау нұсқалары. ORDER BY бөлігі тек осы сөздіктен алынады, # сондықтан SQL-инъекциядан қорғалған. SORT_OPTIONS = { "сәйкестік": "match_percent DESC, v.salary DESC", "жалақы_жоғары": "v.salary DESC, match_percent DESC", "жалақы_төмен": "v.salary ASC, match_percent DESC", "тәжірибе_аз": "v.experience_years ASC, match_percent DESC", "тәжірибе_көп": "v.experience_years DESC, match_percent DESC", "атауы": "v.title ASC", } def recommend_vacancies(user_id, sort_by="сәйкестік", same_city=False, min_salary=0, max_experience=None, top_n=5): """Пайдаланушыға сәйкес вакансияларды ұсынады. sort_by - сұрыптау түрі (SORT_OPTIONS кілттерінің бірі) same_city - True болса, тек пайдаланушының қаласындағы вакансиялар min_salary - ең төменгі жалақы max_experience - вакансияның тәжірибе талабының жоғарғы шегі (None болса, шектеусіз) top_n - нәтижедегі вакансиялар саны """ if sort_by not in SORT_OPTIONS: raise ValueError( f"Қате сұрыптау түрі. Мүмкін нұсқалар: {list(SORT_OPTIONS)}") exp_limit = 999 if max_experience is None else max_experience query = f""" SELECT v.title, c.company_name, v.city, v.salary, v.experience_years, COUNT(vs.skill_id) AS required_skills, COUNT(us.skill_id) AS matched_skills, ROUND(100.0 * COUNT(us.skill_id) / COUNT(vs.skill_id), 1) AS match_percent FROM Vacancies v JOIN Companies c ON v.company_id = c.company_id JOIN Vacancy_Skills vs ON v.vacancy_id = vs.vacancy_id LEFT JOIN User_Skills us ON us.skill_id = vs.skill_id AND us.user_id = ? WHERE (? = 0 OR v.city = (SELECT city FROM Users WHERE user_id = ?)) AND v.salary >= ? AND v.experience_years <= ? GROUP BY v.vacancy_id, v.title, c.company_name, v.city, v.salary, v.experience_years HAVING matched_skills > 0 ORDER BY {SORT_OPTIONS[sort_by]} LIMIT ? """ params = (user_id, int(same_city), user_id, min_salary, exp_limit, top_n) df = pd.read_sql_query(query, conn, params=params) df.columns = ["Вакансия", "Компания", "Қала", "Жалақы", "Тәжірибе (жыл)", "Қажетті дағдылар", "Сәйкес дағдылар", "Сәйкестік (%)"] return df def missing_skills(user_id): """Пайдаланушыда жетіспейтін дағдыларды вакансия бойынша көрсетеді (тек пайдаланушыға ішінара сәйкес вакансиялар үшін).""" query = """ SELECT v.title AS "Вакансия", s.skill_name AS "Жетіспейтін дағды", vs.required_level AS "Қажетті деңгей" FROM Vacancies v JOIN Vacancy_Skills vs ON v.vacancy_id = vs.vacancy_id JOIN Skills s ON s.skill_id = vs.skill_id WHERE vs.skill_id NOT IN ( SELECT skill_id FROM User_Skills WHERE user_id = ? ) AND v.vacancy_id IN ( SELECT vs2.vacancy_id FROM Vacancy_Skills vs2 JOIN User_Skills us2 ON us2.skill_id = vs2.skill_id WHERE us2.user_id = ? ) ORDER BY v.title """ return pd.read_sql_query(query, conn, params=(user_id, user_id)) def matched_skills(user_id): """Қандай дағдылар сәйкес келгенін көрсетеді (ұсыныстың себебі).""" query = """ SELECT v.title AS "Вакансия", s.skill_name AS "Сәйкес дағды", us.skill_level AS "Сіздің деңгейіңіз", vs.required_level AS "Қажетті деңгей" FROM Vacancies v JOIN Vacancy_Skills vs ON v.vacancy_id = vs.vacancy_id JOIN User_Skills us ON us.skill_id = vs.skill_id AND us.user_id = ? JOIN Skills s ON s.skill_id = vs.skill_id ORDER BY v.title """ return pd.read_sql_query(query, conn, params=(user_id,)) def user_name(user_id): row = cursor.execute( "SELECT full_name FROM Users WHERE user_id = ?", (user_id,) ).fetchone() return row[0] if row else None # ===================================================================== # 5. ҰСЫНЫСТАРДЫ КӨРСЕТУ (мысалдар) # ===================================================================== USER_ID = 1 print("#" * 70) print(f" {user_name(USER_ID)} үшін ұсыныстар") print("#" * 70, "\n") print("1) Сәйкестік бойынша (әдепкі):") print(recommend_vacancies(USER_ID).to_string(index=False), "\n") print("2) Жалақысы жоғарыдан төменге (жоғары жалақылар):") print(recommend_vacancies(USER_ID, sort_by="жалақы_жоғары") .to_string(index=False), "\n") print("3) Жалақысы төменнен жоғарыға:") print(recommend_vacancies(USER_ID, sort_by="жалақы_төмен") .to_string(index=False), "\n") print("4) Тәжірибе талабы аз вакансиялар бірінші:") print(recommend_vacancies(USER_ID, sort_by="тәжірибе_аз") .to_string(index=False), "\n") print("5) Тек өз қаласында, жалақы 400 000-нан жоғары:") print(recommend_vacancies(USER_ID, sort_by="жалақы_жоғары", same_city=True, min_salary=400000) .to_string(index=False), "\n") print("6) Тәжірибе талабы 1 жылдан аспайтын вакансиялар:") print(recommend_vacancies(USER_ID, max_experience=1) .to_string(index=False), "\n") print("7) Сәйкес дағдылар (ұсыныс себебі):") print(matched_skills(USER_ID).to_string(index=False), "\n") print("8) Жетіспейтін дағдылар:") print(missing_skills(USER_ID).to_string(index=False), "\n") # ===================================================================== # 6. ИНТЕРАКТИВТІ МӘЗІР (қаласаңыз, төмендегі жолды ашыңыз) # ===================================================================== def menu(): keys = list(SORT_OPTIONS) while True: print("\n--- ВАКАНСИЯ ҰСЫНУ ЖҮЙЕСІ ---") for u in cursor.execute( "SELECT user_id, full_name, city FROM Users").fetchall(): print(f" {u[0]}. {u[1]} ({u[2]})") raw = input("Пайдаланушы ID (шығу үшін 0): ").strip() if raw == "0": break if not raw.isdigit() or user_name(int(raw)) is None: print("Мұндай пайдаланушы жоқ.") continue uid = int(raw) print("\nСұрыптау түрі:") for i, k in enumerate(keys, 1): print(f" {i}. {k}") s = input("Нөмірін таңдаңыз (Enter = сәйкестік): ").strip() sort_by = keys[int(s) - 1] if s.isdigit() and 1 <= int(s) <= len(keys) \ else "сәйкестік" city = input("Тек өз қаласында ма?
